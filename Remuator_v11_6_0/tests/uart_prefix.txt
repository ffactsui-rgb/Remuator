struct String : std::string {
  using std::string::string;
  String() = default;
  String(const std::string &s) : std::string(s) {}
  template<class T, typename std::enable_if<std::is_arithmetic<T>::value, int>::type=0>
  String(T n) : std::string(std::to_string(n)) {}
  long toInt() const {
    try {
      return std::stol(*this);
    } catch (...) {
      return 0;
    }
  }
  float toFloat() const {
    try {
      return std::stof(*this);
    } catch (...) {
      return 0;
    }
  }
};
struct Preferences {
  std::map<std::string, double> values;
#define PREF(name, type)                                                                           \
  type get##name(const char *k, type d) { return values.count(k) ? type(values[k]) : d; }          \
  size_t put##name(const char *k, type v) {                                                        \
    values[k] = v;                                                                                 \
    return sizeof(v);                                                                              \
  }
  PREF(UInt, uint32_t)
  PREF(UChar, uint8_t) PREF(Char, int8_t) PREF(UShort, uint16_t) PREF(Float, float) PREF(Bool, bool)
#undef PREF
} prefs;
struct Server {
  std::map<std::string, std::string> args;
  int code = 0;
  String response;
  bool hasArg(const char *k) { return args.count(k); }
  String arg(const char *k) { return args[k]; }
  void send(int c, const char *, const String &r) {
    code = c;
    response = r;
  }
} server;

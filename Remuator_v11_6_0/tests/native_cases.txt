void rebootAt(int64_t epoch) {
  mono=0;wall=epoch;motor_Naklon={};motor_Povorot={};
  programRunning=programPaused=false;serialMoveNaklonActive=serialMovePovorotActive=false;
  recoveryRequired=recoveryAutoPending=recoveryMotionUncertain=false;
  executionReadError=false;executionMessage="";
  restoreExecutionAfterBoot();
}
void finishAxes() {
  motor_Naklon.complete();executeProgram();
  advance(1);executeProgram();executeProgram();
  motor_Povorot.complete();executeProgram();
}

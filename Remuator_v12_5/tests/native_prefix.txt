struct FakeFS {
  std::map<std::string,std::vector<uint8_t>> files;
  std::set<std::string> dirs;
  bool fail=false;
  bool exists(const char*p){return files.count(p)||dirs.count(p);}
  bool mkdir(const char*p){dirs.insert(p);return !fail;}
  File open(const char*p,const char*mode){
    if(dirs.count(p))return {nullptr,true,0,&fail};
    if(mode[0]=='r'&&!files.count(p))return {};
    if(mode[0]=='w') files[p].clear();
    auto &b=files[p];return {&b,false,mode[0]=='a'?b.size():0,&fail};
  }
  bool rename(const char*a,const char*b){if(fail||!files.count(a))return false;files[b]=files[a];files.erase(a);return true;}
} LittleFS;
